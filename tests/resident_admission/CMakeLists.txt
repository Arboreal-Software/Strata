cmake_minimum_required(VERSION 3.18)
project(resident_admission_fixture LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
get_filename_component(source_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(resident_admission_test test.cpp "${source_root}/src/core/window_budget.cpp")
target_compile_features(resident_admission_test PRIVATE cxx_std_20)
target_compile_options(resident_admission_test PRIVATE -UNDEBUG)
target_include_directories(resident_admission_test PRIVATE "${source_root}/include")
target_link_libraries(resident_admission_test PRIVATE Threads::Threads)
add_test(NAME resident_admission_source COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/check_source.py)
add_test(NAME resident_admission_cases COMMAND resident_admission_test)
set_tests_properties(resident_admission_source PROPERTIES FIXTURES_SETUP resident_admission_source)
set_tests_properties(resident_admission_cases PROPERTIES FIXTURES_REQUIRED resident_admission_source TIMEOUT 30)
