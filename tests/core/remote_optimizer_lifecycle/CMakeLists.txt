find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(_lifecycle_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_custom_target(remote_optimizer_lifecycle_freshness COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/check_source.py")
add_executable(remote_optimizer_lifecycle_test test.cpp)
add_executable(remote_optimizer_sycl_stub_host_test sycl-stub-test.cpp "${_lifecycle_root}/sycl/src/core/remote_expert_opt_stub.cpp")
foreach(t remote_optimizer_lifecycle_test remote_optimizer_sycl_stub_host_test)
 target_compile_features(${t} PRIVATE cxx_std_17)
 target_compile_options(${t} PRIVATE -O1 -g -UNDEBUG -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(${t} PRIVATE -fsanitize=address,undefined)
 target_include_directories(${t} PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${_lifecycle_root}/include")
 add_dependencies(${t} remote_optimizer_lifecycle_freshness)
 add_test(NAME ${t} COMMAND ${t})
 set_tests_properties(${t} PROPERTIES TIMEOUT 90 ENVIRONMENT "ASAN_OPTIONS=detect_leaks=1:abort_on_error=1")
endforeach()
