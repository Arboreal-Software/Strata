cmake_minimum_required(VERSION 3.18)
project(all_resident_remote_combine_fixture LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(source_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
foreach(mode before fixed)
 add_executable(all_resident_remote_combine_${mode} test.cpp)
 target_compile_features(all_resident_remote_combine_${mode} PRIVATE cxx_std_17)
 target_include_directories(all_resident_remote_combine_${mode} PRIVATE "${source_root}/include")
 target_compile_options(all_resident_remote_combine_${mode} PRIVATE -UNDEBUG)
endforeach()
target_compile_definitions(all_resident_remote_combine_before PRIVATE BEFORE=1)
add_test(NAME all_resident_remote_combine_freshness COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/check_freshness.py)
add_test(NAME all_resident_remote_combine_expected_red COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/expect_red.py $<TARGET_FILE:all_resident_remote_combine_before>)
add_test(NAME all_resident_remote_combine_fixed COMMAND all_resident_remote_combine_fixed)
set_tests_properties(all_resident_remote_combine_freshness PROPERTIES FIXTURES_SETUP all_resident_source)
set_tests_properties(all_resident_remote_combine_expected_red all_resident_remote_combine_fixed PROPERTIES FIXTURES_REQUIRED all_resident_source TIMEOUT 25)
