cmake_minimum_required(VERSION 3.20)
project(canonical_admission_consumers LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
get_filename_component(source_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_test(NAME canonical_consumer_freshness COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/check_source.py)
set_tests_properties(canonical_consumer_freshness PROPERTIES FIXTURES_SETUP canonical_consumer_source)
foreach(variant IN ITEMS fixed range batch pool)
  set(target canonical_consumer_${variant})
  add_executable(${target} test.cpp "${source_root}/src/core/window_budget.cpp")
  target_compile_features(${target} PRIVATE cxx_std_20)
  target_include_directories(${target} PRIVATE "${source_root}/include")
  target_link_libraries(${target} PRIVATE Threads::Threads)
  if(NOT variant STREQUAL "fixed")
    string(TOUPPER "${variant}" upper_variant)
    target_compile_definitions(${target} PRIVATE CANONICAL_OLD_${upper_variant})
  endif()
  add_test(NAME ${target} COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/supervise.py ${variant} $<TARGET_FILE:${target}>)
  set_tests_properties(${target} PROPERTIES FIXTURES_REQUIRED canonical_consumer_source TIMEOUT 15)
endforeach()
