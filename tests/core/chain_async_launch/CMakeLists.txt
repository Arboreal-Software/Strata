# Standalone exact-source CPU fixture. No CUDA runtime/engine or GPU linkage.
cmake_minimum_required(VERSION 3.18)
project(chain_async_launch_fixture LANGUAGES CXX)
enable_testing()
find_package(Threads REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(source_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
foreach(mode before fixed)
 add_executable(chain_async_launch_${mode} test.cpp)
 target_compile_features(chain_async_launch_${mode} PRIVATE cxx_std_17)
 target_include_directories(chain_async_launch_${mode} PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${source_root}/include")
 target_compile_options(chain_async_launch_${mode} PRIVATE -UNDEBUG)
 target_link_libraries(chain_async_launch_${mode} PRIVATE Threads::Threads)
endforeach()
target_compile_definitions(chain_async_launch_before PRIVATE BEFORE=1)
add_test(NAME chain_async_launch_freshness COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/check_freshness.py)
add_test(NAME chain_async_launch_expected_red COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/expect_red.py $<TARGET_FILE:chain_async_launch_before>)
add_test(NAME chain_async_launch_fixed COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/run_fixed.py $<TARGET_FILE:chain_async_launch_fixed>)
set_tests_properties(chain_async_launch_freshness PROPERTIES FIXTURES_SETUP chain_launch_source)
set_tests_properties(chain_async_launch_expected_red chain_async_launch_fixed PROPERTIES FIXTURES_REQUIRED chain_launch_source TIMEOUT 30)
