cmake_minimum_required(VERSION 3.18)
project(checked_device_fixture LANGUAGES CXX)
find_package(Threads REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()
add_executable(checked_device_fixture fixture/test.cpp)
target_compile_features(checked_device_fixture PRIVATE cxx_std_17)
target_include_directories(checked_device_fixture PRIVATE fixture source/include)
target_link_libraries(checked_device_fixture PRIVATE Threads::Threads)
add_test(NAME checked_device_source_freshness COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/fixture/check_source.py)
add_test(NAME checked_device_eleven_modes COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/fixture/supervise.py $<TARGET_FILE:checked_device_fixture>)
set_tests_properties(checked_device_source_freshness PROPERTIES TIMEOUT 20)
set_tests_properties(checked_device_eleven_modes PROPERTIES TIMEOUT 90 DEPENDS checked_device_source_freshness)
