cmake_minimum_required(VERSION 3.18)
project(verifier_completion_gate_fixture LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
get_filename_component(source_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_executable(verifier_completion_gate_test test.cpp "${source_root}/src/core/window_budget.cpp")
target_compile_features(verifier_completion_gate_test PRIVATE cxx_std_20)
target_compile_options(verifier_completion_gate_test PRIVATE -UNDEBUG)
target_include_directories(verifier_completion_gate_test PRIVATE "${source_root}/include")
target_link_libraries(verifier_completion_gate_test PRIVATE Threads::Threads)
add_test(NAME verifier_completion_gate_freshness COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/check_source.py)
add_test(NAME verifier_completion_gate_cases COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/supervise.py $<TARGET_FILE:verifier_completion_gate_test>)
set_tests_properties(verifier_completion_gate_freshness PROPERTIES FIXTURES_SETUP completion_gate_source)
set_tests_properties(verifier_completion_gate_cases PROPERTIES FIXTURES_REQUIRED completion_gate_source TIMEOUT 25)
