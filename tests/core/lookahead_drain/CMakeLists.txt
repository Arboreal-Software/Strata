cmake_minimum_required(VERSION 3.20)
project(lookahead_drain_fixture LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
add_executable(lookahead_drain_test test.cpp)
target_compile_features(lookahead_drain_test PRIVATE cxx_std_20)
target_compile_options(lookahead_drain_test PRIVATE -Wall -Wextra -fsanitize=address,undefined -fno-omit-frame-pointer)
target_link_options(lookahead_drain_test PRIVATE -fsanitize=address,undefined)
target_link_libraries(lookahead_drain_test PRIVATE Threads::Threads)
add_test(NAME lookahead_drain_freshness COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/extract.py)
add_test(NAME lookahead_drain_cases COMMAND lookahead_drain_test)
set_tests_properties(lookahead_drain_cases PROPERTIES DEPENDS lookahead_drain_freshness TIMEOUT 20)
add_executable(lookahead_notify_one_control test.cpp)
target_compile_definitions(lookahead_notify_one_control PRIVATE LOOKAHEAD_OLD_NOTIFY_ONE=1)
target_compile_features(lookahead_notify_one_control PRIVATE cxx_std_20)
target_compile_options(lookahead_notify_one_control PRIVATE -Wall -Wextra -fsanitize=address,undefined -fno-omit-frame-pointer)
target_link_options(lookahead_notify_one_control PRIVATE -fsanitize=address,undefined)
target_link_libraries(lookahead_notify_one_control PRIVATE Threads::Threads)
add_test(NAME lookahead_previous_notify_one_control COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/expect_notify_red.py $<TARGET_FILE:lookahead_notify_one_control>)
set_tests_properties(lookahead_previous_notify_one_control PROPERTIES DEPENDS lookahead_drain_freshness TIMEOUT 20)
