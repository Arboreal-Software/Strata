# Standalone CPU fixture; no CUDA/device runtime and no production engine linkage.
cmake_minimum_required(VERSION 3.18)
project(verifier_serial_copy_fixture LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(verifier_serial_copy_before test.cpp)
target_compile_definitions(verifier_serial_copy_before PRIVATE BEFORE=1)
add_executable(verifier_serial_copy_fixed test.cpp)
foreach(target verifier_serial_copy_before verifier_serial_copy_fixed)
  target_compile_features(${target} PRIVATE cxx_std_17)
endforeach()
add_test(NAME verifier_serial_copy_freshness COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/check_freshness.py)
add_test(NAME verifier_serial_copy_expected_red COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/expect_red.py $<TARGET_FILE:verifier_serial_copy_before>)
add_test(NAME verifier_serial_copy_fixed_test COMMAND verifier_serial_copy_fixed)
set_tests_properties(verifier_serial_copy_freshness PROPERTIES FIXTURES_SETUP serial_copy_source)
set_tests_properties(verifier_serial_copy_expected_red verifier_serial_copy_fixed_test PROPERTIES FIXTURES_REQUIRED serial_copy_source TIMEOUT 45)
